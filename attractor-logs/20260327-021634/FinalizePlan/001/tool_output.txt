 131 |     {
 132 |         // Compute entirely in Int128 so that boundary dates with timezone offsets
 133 |         // are handled correctly. The offset must be applied before range-checking,
 134 |         // otherwise a pre-offset value that exceeds long range would be rejected
 135 |         // even if the final UTC result fits (e.g. +292278994-08-17T08:12:55.807+0100).
 136 |         long days = GregorianDateTime.DaysFromCivil(year, month, day);
 137 |         Int128 totalMs =
 138 |             (Int128)days * CedarConsts.MillisPerDay
 139 |             + hour * (long)CedarConsts.MillisPerHour
 140 |             + minute * (long)CedarConsts.MillisPerMinute
 141 |             + second * (long)CedarConsts.MillisPerSecond
 142 |             + millisecond
 143 |             - offsetMilliseconds;
 144 |

 146 |         {
 147 |             throw new FormatException("Timestamp is out of range.");
 148 |         }

 205 |     {
 206 |         cursor++;
 207 | 
 208 |         if (value.Length != cursor + 4)
 209 |         {
 210 |             throw new FormatException("Time zone offsets must use the +hhmm or -hhmm form.");
 211 |         }
 212 | 
 213 |         int hours = ParseDigits(value, ref cursor, 2, "time zone hour");
 214 |         int minutes = ParseDigits(value, ref cursor, 2, "time zone minute");
 215 | 
 216 |         if (hours > 23)
 217 |         {
 218 |             throw new FormatException("Time zone offset hours are greater than 23.");
 219 |         }
 220 | 
 221 |         if (minutes > 59)
 222 |         {
 223 |             throw new FormatException("Time zone offset minutes are greater than 59.");
 224 |         }
 225 | 
 226 |         long magnitude = (hours * CedarConsts.MillisPerHour) + (minutes * CedarConsts.MillisPerMinute);
 227 |         return sign * magnitude;
 228 |     }
