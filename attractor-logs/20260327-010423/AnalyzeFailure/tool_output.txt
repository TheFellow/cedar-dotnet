  10 |     public static CedarDatetime Parse(string value)
  11 |     {
  12 |         ArgumentNullException.ThrowIfNull(value);
  13 | 
  14 |         if (value.Length == 0)
  15 |         {
  16 |             throw new FormatException("Datetime values must not be empty.");
  17 |         }
  18 | 
  19 |         int cursor = 0;
  20 |         int yearSign = 1;
  21 |         int yearDigits = 4;
  22 | 
  23 |         if (value[cursor] is '+' or '-')
  24 |         {
  25 |             yearDigits = 9;
  26 |             if (value[cursor] == '-')
  27 |             {
  28 |                 yearSign = -1;
  29 |             }
  30 | 
  31 |             cursor++;
  32 |         }
  33 | 
  34 |         int yearMagnitude = ParseDigits(value, ref cursor, yearDigits, "year");
  35 |         int year = checked(yearSign * yearMagnitude);
  36 | 
  37 |         ExpectCharacter(value, ref cursor, '-');
  38 |         int month = ParseDigits(value, ref cursor, 2, "month");
  39 |         ExpectCharacter(value, ref cursor, '-');
  40 |         int day = ParseDigits(value, ref cursor, 2, "day");
  41 | 
  42 |         ValidateDate(year, month, day);
  43 | 
  44 |         if (cursor == value.Length)
  45 |         {
  46 |             return CreateFromParts(year, month, day, 0, 0, 0, 0, 0);
  47 |         }
  48 | 
  49 |         ExpectCharacter(value, ref cursor, 'T');
  50 |         int hour = ParseDigits(value, ref cursor, 2, "hour");
  51 |         ExpectCharacter(value, ref cursor, ':');
  52 |         int minute = ParseDigits(value, ref cursor, 2, "minute");
  53 |         ExpectCharacter(value, ref cursor, ':');
  54 |         int second = ParseDigits(value, ref cursor, 2, "second");
  55 | 
  56 |         if (hour > 23)
  57 |         {
  58 |             throw new FormatException("Hour is greater than 23.");
  59 |         }
  60 | 
  61 |         if (minute > 59)
  62 |         {
  63 |             throw new FormatException("Minute is greater than 59.");
  64 |         }
  65 | 
  66 |         if (second > 59)
  67 |         {
  68 |             throw new FormatException("Second is greater than 59.");
  69 |         }
  70 | 
  71 |         int milliseconds = 0;
  72 | 
  73 |         if (cursor < value.Length && value[cursor] == '.')
  74 |         {
  75 |             cursor++;
  76 |             milliseconds = ParseDigits(value, ref cursor, 3, "millisecond");
  77 |         }
  78 | 
  79 |         if (cursor == value.Length)
  80 |         {
  81 |             throw new FormatException("Datetime values must include a time zone designator.");
  82 |         }
  83 | 
  84 |         long offsetMilliseconds = value[cursor] switch
  85 |         {
  86 |             'Z' => ReadUtcTrailer(value, ref cursor),
  87 |             '+' => ReadOffset(value, ref cursor, 1),
  88 |             '-' => ReadOffset(value, ref cursor, -1),
  89 |             _ => throw new FormatException("Invalid time zone designator.")
  90 |         };
  91 | 
  92 |         if (cursor != value.Length)
  93 |         {
  94 |             throw new FormatException("Unexpected trailing characters after the time zone designator.");
  95 |         }
  96 | 
  97 |         return CreateFromParts(year, month, day, hour, minute, second, milliseconds, offsetMilliseconds);

 165 |     private static int ParseDigits(string value, ref int cursor, int length, string component)
 166 |     {
 167 |         if (value.Length < cursor + length)
 168 |         {
 169 |             throw new FormatException($"Invalid {component} component.");
 170 |         }
 171 | 
 172 |         int parsed = 0;
 173 |         for (int index = 0; index < length; index++)
 174 |         {
 175 |             char character = value[cursor + index];
 176 |             if (!char.IsAsciiDigit(character))
 177 |             {
 178 |                 throw new FormatException($"Invalid {component} component.");
 179 |             }
 180 | 
 181 |             parsed = checked(parsed * 10 + (character - '0'));
 182 |         }
 183 | 
 184 |         cursor += length;

 201 |         }

 209 |         }

 213 |             throw new FormatException("Time zone offset minutes are greater than 59.");
 214 |         }
