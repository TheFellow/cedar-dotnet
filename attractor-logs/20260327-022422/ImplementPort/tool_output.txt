 174 |     private static GregorianDateTimeParts NormalizeUtcParts(int year, int month, int day, int hour, int minute, int second, int millisecond, long offsetMilliseconds)
 175 |     {
 176 |         long timeOfDayMilliseconds =
 177 |             (hour * (long)CedarConsts.MillisPerHour)
 178 |             + (minute * (long)CedarConsts.MillisPerMinute)
 179 |             + (second * (long)CedarConsts.MillisPerSecond)
 180 |             + millisecond
 181 |             - offsetMilliseconds;
 182 | 
 183 |         int dayAdjustment = 0;
 184 |         if (timeOfDayMilliseconds < 0)
 185 |         {
 186 |             dayAdjustment = -1;
 187 |             timeOfDayMilliseconds += CedarConsts.MillisPerDay;
 188 |         }
 189 |         else if (timeOfDayMilliseconds >= CedarConsts.MillisPerDay)
 190 |         {
 191 |             dayAdjustment = 1;
 192 |             timeOfDayMilliseconds -= CedarConsts.MillisPerDay;
 193 |         }
 194 | 
 195 |         AdjustCivilDate(ref year, ref month, ref day, dayAdjustment);
 196 | 
 197 |         int utcHour = (int)(timeOfDayMilliseconds / CedarConsts.MillisPerHour);
 198 |         timeOfDayMilliseconds %= CedarConsts.MillisPerHour;
 199 | 
 200 |         int utcMinute = (int)(timeOfDayMilliseconds / CedarConsts.MillisPerMinute);
 201 |         timeOfDayMilliseconds %= CedarConsts.MillisPerMinute;
 202 | 
 203 |         int utcSecond = (int)(timeOfDayMilliseconds / CedarConsts.MillisPerSecond);
 204 |         int utcMillisecond = (int)(timeOfDayMilliseconds % CedarConsts.MillisPerSecond);
 205 | 
 206 |         return new GregorianDateTimeParts(year, month, day, utcHour, utcMinute, utcSecond, utcMillisecond);
 207 |     }

 209 |     private static void AdjustCivilDate(ref int year, ref int month, ref int day, int dayAdjustment)
 210 |     {
 211 |         if (dayAdjustment == 0)
 212 |         {
 213 |             return;
 214 |         }
 215 | 
 216 |         if (dayAdjustment > 0)
 217 |         {
 218 |             day++;
 219 |             int daysInMonth = GregorianDateTime.DaysInMonth(year, month);
 220 |             if (day > daysInMonth)
 221 |             {
 222 |                 day = 1;
 223 |                 month++;
 224 |                 if (month > 12)
 225 |                 {
 226 |                     month = 1;
 227 |                     year = checked(year + 1);
 228 |                 }
 229 |             }
 230 | 
 231 |             return;
 232 |         }
 233 | 
 234 |         day--;
 235 |         if (day > 0)
 236 |         {
 237 |             return;
 238 |         }
 239 | 
 240 |         month--;
 241 |         if (month < 1)
 242 |         {
 243 |             month = 12;
 244 |             year = checked(year - 1);
 245 |         }
 246 | 
 247 |         day = GregorianDateTime.DaysInMonth(year, month);
 248 |     }

 301 |     private static int ParseDigits(string value, ref int cursor, int length, string component)
 302 |     {
 303 |         if (value.Length < cursor + length)
 304 |         {
 305 |             throw new FormatException($"Invalid {component} component.");
 306 |         }
 307 | 
 308 |         int parsed = 0;
 309 |         for (int index = 0; index < length; index++)
 310 |         {
 311 |             char character = value[cursor + index];
 312 |             if (!char.IsAsciiDigit(character))
 313 |             {
 314 |                 throw new FormatException($"Invalid {component} component.");
 315 |             }
 316 | 
 317 |             parsed = checked(parsed * 10 + (character - '0'));
 318 |         }
 319 | 
 320 |         cursor += length;
 321 |         return parsed;
 322 |     }

 330 |     private static long ReadOffset(string value, ref int cursor, int sign)
 331 |     {
 332 |         cursor++;
 333 | 
 334 |         if (value.Length != cursor + 4)
 335 |         {
 336 |             throw new FormatException("Time zone offsets must use the +hhmm or -hhmm form.");
 337 |         }
 338 | 
 339 |         int hours = ParseDigits(value, ref cursor, 2, "time zone hour");
 340 |         int minutes = ParseDigits(value, ref cursor, 2, "time zone minute");
 341 | 
 342 |         if (hours > 23)
 343 |         {
 344 |             throw new FormatException("Time zone offset hours are greater than 23.");
 345 |         }
 346 | 
 347 |         if (minutes > 59)
 348 |         {
 349 |             throw new FormatException("Time zone offset minutes are greater than 59.");
 350 |         }
 351 | 
 352 |         long magnitude = (hours * CedarConsts.MillisPerHour) + (minutes * CedarConsts.MillisPerMinute);
 353 |         return sign * magnitude;
 354 |     }

 361 |         }

 365 |             throw new FormatException("Day is greater than 31.");
 366 |         }
 367 |

 369 |         if (day > daysInMonth)
 370 |         {
 371 |             throw new FormatException("Date is invalid.");
 372 |         }

  77 | // ParseDatetime returns a Cedar datetime when the argument provided
  78 | // represents a compatible datetime or an error
  79 | //
  80 | // Cedar RFC 80 defines valid datetime strings as one of:
  81 | //
  82 | // - "YYYY-MM-DD" (date only, with implied time 00:00:00, UTC)
  83 | // - "YYYY-MM-DDThh:mm:ssZ" (date and time, UTC)
  84 | // - "YYYY-MM-DDThh:mm:ss.SSSZ" (date and time with millisecond, UTC)
  85 | // - "YYYY-MM-DDThh:mm:ss(+/-)hhmm" (date and time, time zone offset)
  86 | // - "YYYY-MM-DDThh:mm:ss.SSS(+/-)hhmm" (date and time with millisecond, time zone offset)
  87 | //
  88 | // Cedar RFC 110 extends this with ISO 8601 expanded year format:
  89 | //
  90 | // - "(+/-)YYYYYYYYY-MM-DD" (9-digit year, date only)
  91 | // - "(+/-)YYYYYYYYY-MM-DDThh:mm:ssZ" (9-digit year, date and time, UTC)
  92 | // - "(+/-)YYYYYYYYY-MM-DDThh:mm:ss.SSSZ" (9-digit year with millisecond, UTC)
  93 | // - "(+/-)YYYYYYYYY-MM-DDThh:mm:ss(+/-)hhmm" (9-digit year with time zone offset)
  94 | // - "(+/-)YYYYYYYYY-MM-DDThh:mm:ss.SSS(+/-)hhmm" (9-digit year with millisecond and offset)
  95 | func ParseDatetime(s string) (Datetime, error) {
  96 | 	var (
  97 | 		year                                    int
  98 | 		month, day, hour, minute, second, milli uint
  99 | 		offset                                  time.Duration
 100 | 	)
 101 | 
 102 | 	if len(s) == 0 {
 103 | 		return Datetime{}, fmt.Errorf("%w: unexpected EOF", errDatetime)
 104 | 	}
 105 | 
 106 | 	// Check if this is an expanded year format (starts with + or -)
 107 | 	yearSign := 1
 108 | 	yearLength := 4
 109 | 	yearMax := uint(9999)
 110 | 	if s[0] == '+' || s[0] == '-' {
 111 | 		yearLength = 9
 112 | 		yearMax = 999999999
 113 | 		if s[0] == '-' {
 114 | 			yearSign = -1
 115 | 		}
 116 | 		s = s[1:]
 117 | 	} else if !unicode.IsDigit(rune(s[0])) {
 118 | 		return Datetime{}, fmt.Errorf("%w: invalid year", errDatetime)
 119 | 	}
 120 | 
 121 | 	absYear, s, err := parseUint(s[0:], yearLength, yearMax, "year")
 122 | 	if err != nil {
 123 | 		return Datetime{}, err
 124 | 	}
 125 | 	year = int(absYear) * yearSign
 126 | 
 127 | 	if s, err = expectChar(s, '-'); err != nil {
 128 | 		return Datetime{}, err
 129 | 	}
 130 | 
 131 | 	if month, s, err = parseUint(s, 2, 12, "month"); err != nil {
 132 | 		return Datetime{}, err
 133 | 	}
 134 | 
 135 | 	if s, err = expectChar(s, '-'); err != nil {
 136 | 		return Datetime{}, err
 137 | 	}
 138 | 
 139 | 	if day, s, err = parseUint(s, 2, 31, "day"); err != nil {
 140 | 		return Datetime{}, err
 141 | 	}
 142 | 
 143 | 	if err = checkValidDay(year, month, day); err != nil {
 144 | 		return Datetime{}, err
 145 | 	}
 146 | 
 147 | 	if len(s) == 0 {
 148 | 		return Datetime{time.Date(year, time.Month(month), int(day), 0, 0, 0, 0, time.UTC).UnixMilli()}, nil
 149 | 	}
 150 | 
 151 | 	if s, err = expectChar(s, 'T'); err != nil {
 152 | 		return Datetime{}, err
 153 | 	}
 154 | 
 155 | 	if hour, s, err = parseUint(s, 2, 23, "hour"); err != nil {
 156 | 		return Datetime{}, err
 157 | 	}
 158 | 
 159 | 	if s, err = expectChar(s, ':'); err != nil {
 160 | 		return Datetime{}, err
 161 | 	}
 162 | 
 163 | 	if minute, s, err = parseUint(s, 2, 59, "minute"); err != nil {
 164 | 		return Datetime{}, err
 165 | 	}
 166 | 
 167 | 	if s, err = expectChar(s, ':'); err != nil {
 168 | 		return Datetime{}, err
 169 | 	}
 170 | 
 171 | 	if second, s, err = parseUint(s, 2, 59, "second"); err != nil {
 172 | 		return Datetime{}, err
 173 | 	}
 174 | 
 175 | 	if len(s) == 0 {
 176 | 		return Datetime{}, fmt.Errorf("%w: unexpected EOF", errDatetime)
 177 | 	}
 178 | 
 179 | 	// Parse optional milliseconds
 180 | 	if s[0] == '.' {
 181 | 		milli, s, err = parseUint(s[1:], 3, 999, "millisecond")
 182 | 		if err != nil {
 183 | 			return Datetime{}, err
 184 | 		}
 185 | 	}
 186 | 
 187 | 	if len(s) == 0 {
 188 | 		return Datetime{}, fmt.Errorf("%w: unexpected EOF", errDatetime)
 189 | 	}
 190 | 
 191 | 	switch s[0] {
 192 | 	case 'Z':
 193 | 		s = s[1:]
 194 | 	case '+', '-':
 195 | 		sign := 1
 196 | 		if s[0] == '-' {
 197 | 			sign = -1
 198 | 		}
 199 | 		s = s[1:]
 200 |
